### 2.2 简单双均线策略

```python
from pandas import DataFrame
import talib.abstract as ta
from freqtrade.strategy import IStrategy

class SimpleMAStrategy(IStrategy):
    """
    简单的双均线交叉策略
    
    买入信号：快线上穿慢线（金叉）
    卖出信号：快线下穿慢线（死叉）
    """
    
    # ========== 基础配置 ==========
    INTERFACE_VERSION = 3
    
    # 时间周期
    timeframe = '5m'
    
    # 止损：亏损10%就卖
    stoploss = -0.10
    
    # 盈利目标
    minimal_roi = {
        "0": 0.10,   # 任何时候盈利10%就卖
        "30": 0.05,  # 30分钟后盈利5%就卖
        "60": 0.02,  # 60分钟后盈利2%就卖
        "120": 0.01  # 120分钟后盈利1%就卖
    }
    
    # 需要的历史K线数量
    startup_candle_count = 100
    
    # ========== 指标计算 ==========
    def populate_indicators(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        """
        计算技术指标
        """
        # 快速均线（10周期）
        dataframe['ma_fast'] = ta.SMA(dataframe, timeperiod=10)
        
        # 慢速均线（50周期）
        dataframe['ma_slow'] = ta.SMA(dataframe, timeperiod=50)
        
        # RSI - 用于过滤极端情况
        dataframe['rsi'] = ta.RSI(dataframe, timeperiod=14)
        
        return dataframe
    
    # ========== 买入信号 ==========
    def populate_entry_trend(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        """
        生成买入信号
        """
        dataframe.loc[
            (
                # 条件1：快线上穿慢线（金叉）
                # 当前快线 > 慢线，但前一根快线 <= 慢线
                (dataframe['ma_fast'] > dataframe['ma_slow']) &
                (dataframe['ma_fast'].shift(1) <= dataframe['ma_slow'].shift(1)) &
                
                # 条件2：RSI不能太高（避免追高）
                (dataframe['rsi'] < 70) &
                
                # 条件3：有成交量
                (dataframe['volume'] > 0)
            ),
            'enter_long'  # 标记买入
        ] = 1
        
        return dataframe
    
    # ========== 卖出信号 ==========
    def populate_exit_trend(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        """
        生成卖出信号
        """
        dataframe.loc[
            (
                # 条件：快线下穿慢线（死叉）
                (dataframe['ma_fast'] < dataframe['ma_slow']) &
                (dataframe['ma_fast'].shift(1) >= dataframe['ma_slow'].shift(1))
            ),
            'exit_long'  # 标记卖出
        ] = 1
        
        return dataframe
```

### 4.1 多指标组合策略

```python
class MultiIndicatorStrategy(IStrategy):
    """
    多指标确认策略
    
    买入条件：
    1. 趋势：均线多头排列
    2. 动量：RSI从超卖区回升
    3. 突破：价格突破布林带中轨
    4. 成交量：成交量放大
    """
    
    INTERFACE_VERSION = 3
    timeframe = '5m'
    stoploss = -0.08
    
    minimal_roi = {
        "0": 0.08,
        "20": 0.04,
        "40": 0.02,
        "80": 0.01
    }
    
    startup_candle_count = 100
    
    def populate_indicators(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        # 趋势指标
        dataframe['ema_short'] = ta.EMA(dataframe, timeperiod=12)
        dataframe['ema_long'] = ta.EMA(dataframe, timeperiod=26)
        
        # 动量指标
        dataframe['rsi'] = ta.RSI(dataframe, timeperiod=14)
        
        # 波动指标
        bollinger = qtpylib.bollinger_bands(dataframe['close'], window=20, stds=2)
        dataframe['bb_lower'] = bollinger['lower']
        dataframe['bb_middle'] = bollinger['mid']
        dataframe['bb_upper'] = bollinger['upper']
        
        # 成交量指标
        dataframe['volume_ma'] = dataframe['volume'].rolling(window=20).mean()
        
        return dataframe
    
    def populate_entry_trend(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        dataframe.loc[
            (
                # 条件1：趋势 - 短期均线在长期均线上方
                (dataframe['ema_short'] > dataframe['ema_long']) &
                
                # 条件2：动量 - RSI从超卖区回升（30-50区间）
                (dataframe['rsi'] > 30) &
                (dataframe['rsi'] < 50) &
                (dataframe['rsi'] > dataframe['rsi'].shift(1)) &
                
                # 条件3：突破 - 价格在布林带中轨附近或上方
                (dataframe['close'] > dataframe['bb_middle']) &
                
                # 条件4：成交量 - 成交量大于平均值
                (dataframe['volume'] > dataframe['volume_ma'] * 1.2) &
                
                # 基础条件
                (dataframe['volume'] > 0)
            ),
            'enter_long'
        ] = 1
        
        return dataframe
    
    def populate_exit_trend(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        dataframe.loc[
            (
                # 退出条件1：RSI超买
                (dataframe['rsi'] > 70) |
                
                # 退出条件2：价格触及布林带上轨
                (dataframe['close'] > dataframe['bb_upper']) |
                
                # 退出条件3：均线死叉
                (
                    (dataframe['ema_short'] < dataframe['ema_long']) &
                    (dataframe['ema_short'].shift(1) >= dataframe['ema_long'].shift(1))
                )
            ),
            'exit_long'
        ] = 1
        
        return dataframe
```

### 4.2 自定义指标策略

```python
class CustomIndicatorStrategy(IStrategy):
    """
    使用自定义指标的策略
    """
    
    INTERFACE_VERSION = 3
    timeframe = '15m'
    stoploss = -0.10
    
    minimal_roi = {
        "0": 0.15,
        "30": 0.08,
        "60": 0.04,
        "120": 0.02
    }
    
    startup_candle_count = 200
    
    def populate_indicators(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        # 自定义指标1：价格动量
        dataframe['price_momentum'] = (
            (dataframe['close'] - dataframe['close'].shift(10)) / 
            dataframe['close'].shift(10) * 100
        )
        
        # 自定义指标2：成交量价格趋势（VPT）
        dataframe['vpt'] = (
            (dataframe['close'] - dataframe['close'].shift(1)) / 
            dataframe['close'].shift(1) * dataframe['volume']
        ).cumsum()
        
        # 自定义指标3：真实波幅百分比
        dataframe['atr'] = ta.ATR(dataframe, timeperiod=14)
        dataframe['atr_percent'] = (dataframe['atr'] / dataframe['close']) * 100
        
        # 自定义指标4：支撑阻力位（简化版）
        # 计算最近50根K线的最高价和最低价
        dataframe['resistance'] = dataframe['high'].rolling(window=50).max()
        dataframe['support'] = dataframe['low'].rolling(window=50).min()
        
        # 价格在支撑阻力区间的位置（0-1）
        dataframe['price_position'] = (
            (dataframe['close'] - dataframe['support']) /
            (dataframe['resistance'] - dataframe['support'])
        )
        
        # 标准指标
        dataframe['rsi'] = ta.RSI(dataframe, timeperiod=14)
        
        return dataframe
    
    def populate_entry_trend(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        dataframe.loc[
            (
                # 条件1：价格动量向上
                (dataframe['price_momentum'] > 0) &
                (dataframe['price_momentum'] > dataframe['price_momentum'].shift(1)) &
                
                # 条件2：成交量价格趋势向上
                (dataframe['vpt'] > dataframe['vpt'].shift(5)) &
                
                # 条件3：价格接近支撑位（底部买入）
                (dataframe['price_position'] < 0.3) &
                
                # 条件4：波动率适中（避免在高波动时进场）
                (dataframe['atr_percent'] < 5) &
                
                # 条件5：RSI不在超买区
                (dataframe['rsi'] < 65) &
                
                (dataframe['volume'] > 0)
            ),
            'enter_long'
        ] = 1
        
        return dataframe
    
    def populate_exit_trend(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        dataframe.loc[
            (
                # 退出条件：价格接近阻力位或RSI超买
                (
                    (dataframe['price_position'] > 0.9) |
                    (dataframe['rsi'] > 75)
                )
            ),
            'exit_long'
        ] = 1
        
        return dataframe
    
    # 自定义止损：根据ATR动态调整
    def custom_stoploss(self, pair, trade, current_time, current_rate, 
                       current_profit, **kwargs):
        dataframe, _ = self.dp.get_analyzed_dataframe(pair, self.timeframe)
        last_candle = dataframe.iloc[-1].squeeze()
        
        # 使用2倍ATR作为止损距离
        atr = last_candle['atr']
        stop_distance = 2 * atr
        
        # 计算止损价格
        stop_price = current_rate - stop_distance
        
        # 转换为相对止损比例
        from freqtrade.strategy import stoploss_from_absolute
        return stoploss_from_absolute(stop_price, current_rate)
```

import logging
logger = logging.getLogger(__name__)

class DebugStrategy(IStrategy):
    def populate_entry_trend(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        # 详细记录决策过程
        last_candle = dataframe.iloc[-1]
        
        logger.info(f"=== 分析 {metadata['pair']} ===")
        logger.info(f"价格: {last_candle['close']:.2f}")
        logger.info(f"RSI: {last_candle['rsi']:.2f}")
        logger.info(f"MA Fast: {last_candle['ma_fast']:.2f}")
        logger.info(f"MA Slow: {last_candle['ma_slow']:.2f}")
        
        # 检查每个条件
        cond1 = last_candle['rsi'] < 30
        cond2 = last_candle['ma_fast'] > last_candle['ma_slow']
        
        logger.info(f"条件1 (RSI<30): {cond1}")
        logger.info(f"条件2 (金叉): {cond2}")
        
        dataframe.loc[
            (
                (dataframe['rsi'] < 30) &
                (dataframe['ma_fast'] > dataframe['ma_slow'])

            'enter_long'
        ] = 1
        
        if last_candle['enter_long'] == 1:
            logger.warning(f"!!! 生成买入信号: {metadata['pair']} !!!")

class ValidationStrategy(IStrategy):
    def populate_indicators(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        dataframe['rsi'] = ta.RSI(dataframe)
        
        # 验证数据质量
        assert not dataframe['close'].isnull().any(), "价格数据包含空值"
        assert (dataframe['close'] > 0).all(), "价格必须大于0"
        assert not dataframe['rsi'].isnull().any(), "RSI计算出现空值"
        assert (dataframe['rsi'] >= 0).all() and (dataframe['rsi'] <= 100).all(), \
            "RSI值超出范围(0-100)"
        
        return dataframe
