## 一、代码性能优化

### 1.1 向量化操作

**原则：**尽可能使用 Pandas/NumPy 的向量化操作，避免 Python 循环。

```python
import numpy as np
import pandas as pd

# ❌ 低效：Python 循环（慢100倍）
def calculate_sma_slow(dataframe, period=20):
    result = []
    for i in range(len(dataframe)):
        if i < period:
            result.append(np.nan)
        else:
            result.append(dataframe['close'].iloc[i-period:i].mean())
    return result

# ✅ 高效：向量化操作
def calculate_sma_fast(dataframe, period=20):
    return dataframe['close'].rolling(period).mean()

# 性能对比
# 慢版本：10秒
# 快版本：0.1秒
```

from typing import Tuple

class CachedStrategy(IStrategy):
    
    @lru_cache(maxsize=100)
    def expensive_calculation(self, pair: str, timeframe: str) -> dict:
        """
        昂贵的计算，结果会被缓存
        相同的参数不会重复计算
        """
        # 模拟耗时操作
        import time
        time.sleep(1)

            'pair': pair,
            'timeframe': timeframe,
            'result': 'calculated'

    def populate_indicators(self, dataframe, metadata):
        # 第一次调用：1秒

def optimize_dataframe_memory(dataframe):
    """
    优化 DataFrame 内存占用
    """
    # 将 float64 转为 float32（减少50%内存）
    float_cols = dataframe.select_dtypes(include=['float64']).columns
    dataframe[float_cols] = dataframe[float_cols].astype('float32')
    
    # 将 int64 转为 int32
    int_cols = dataframe.select_dtypes(include=['int64']).columns
    dataframe[int_cols] = dataframe[int_cols].astype('int32')
    
    return dataframe

        self.orderbook_cache = {}
        self.cache_timeout = 60  # 缓存60秒

                # 缓存过期，重新获取
                orderbook = self.dp.orderbook(pair, 1)
                self.orderbook_cache[pair] = {
                    'data': orderbook,
                    'time': current_time
                }

### 4.5 异常处理

```python
class RobustStrategy(IStrategy):
    
    def populate_indicators(self, dataframe, metadata):
        try:
            # 计算指标
            dataframe['rsi'] = ta.RSI(dataframe, timeperiod=14)
            dataframe['ema'] = ta.EMA(dataframe, timeperiod=20)
            
            # 检查结果
            if dataframe['rsi'].isna().all():
                logger.error(f"{metadata['pair']} RSI计算失败，全是NaN")
                # 使用默认值
                dataframe['rsi'] = 50
            
            # 填充NaN
            dataframe = dataframe.ffill().bfill()
            
        except Exception as e:
            logger.error(f"{metadata['pair']} 指标计算异常: {e}")
            # 返回安全的默认值
            dataframe['rsi'] = 50
            dataframe['ema'] = dataframe['close']
        
        return dataframe
    
    def populate_entry_trend(self, dataframe, metadata):
        try:
            # 正常逻辑
            dataframe.loc[
                (
                    (dataframe['rsi'] < 30) &
                    (dataframe['ema'] > dataframe['close'])
                ),
                'enter_long'
            ] = 1
            
        except Exception as e:
            logger.error(f"{metadata['pair']} 信号生成异常: {e}")
            # 发生异常时不生成信号
            dataframe['enter_long'] = 0
        
        return dataframe
```

                alerts.append(f"⏰ {trade['pair']} 持仓 {duration:.1f} 小时")
        
        return alerts
        
    except Exception as e:
        return [f"❌ 监控失败: {e}"]

def main():
    while True:
        alerts = check_health()
        
        if alerts:
            print(f"[{datetime.now()}] 发现 {len(alerts)} 个告警:")
            for alert in alerts:
                print(f"  {alert}")
                # 这里可以发送告警（邮件/短信/Webhook）
        else:
            print(f"[{datetime.now()}] ✓ 一切正常")
        
        time.sleep(CHECK_INTERVAL)

if __name__ == "__main__":
    main()
